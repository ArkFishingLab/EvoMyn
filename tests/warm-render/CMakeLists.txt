cmake_minimum_required(VERSION 3.20)
project(warm_render LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "${REPOSITORY_ROOT}/xiaozhi-firmware/managed_components/lvgl__lvgl" CACHE PATH "LVGL 9 source checkout")
set(ROUND_BOARD_DIR "${REPOSITORY_ROOT}/xiaozhi-firmware/main/boards/round-voice-175c" CACHE PATH "Board sources with generated fonts")
if(NOT EXISTS "${LVGL_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Provide -DLVGL_SOURCE_DIR with the LVGL 9 component source directory")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "Host LVGL configuration" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
set(BOARD "${ROUND_BOARD_DIR}")
file(GLOB FONTS "${BOARD}/warm_font_*.cc" "${BOARD}/warm_digits_*.cc")
add_executable(warm_render render.cpp "${BOARD}/warm_ui.cc" "${BOARD}/planner_ui.cc" "${BOARD}/warm_icons.cc" ${FONTS})
target_include_directories(warm_render PRIVATE "${BOARD}")
target_link_libraries(warm_render PRIVATE lvgl)
target_compile_options(warm_render PRIVATE -UNDEBUG)
